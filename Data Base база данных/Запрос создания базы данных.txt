CREATE VIEW [dbo].[vw_Текущие_остатки] AS
SELECT 
    с.id as Склад_id,
    с.Наименование as Склад,
    п.Артикул,
    п.Наименование as Продукт,
    ISNULL(кп.Наименование, 'Без категории') as Категория,
    п.Единица_измерения,
    ISNULL(ост.Количество, 0) as Количество,
    ISNULL(ост.Цена_за_единицу, 0) as Цена_за_единицу,
    ISNULL(ост.Сумма, 0) as Сумма,
    п.Цена as Цена_последняя_закупка,
    ост.Дата_обновления
FROM Склады с
CROSS JOIN Продукты п
LEFT JOIN Остатки_продуктов ост ON ост.Склад_id = с.id AND ост.Продукт_id = п.id
LEFT JOIN Категории_продуктов кп ON п.Категория_id = кп.id
WHERE п.Активен = 1 AND с.Активен = 1;

CREATE VIEW [dbo].[vw_Требования_накладные] AS
SELECT 
    тр.id,
    тр.Номер,
    д.Дата_документа,
    м.Дата as Дата_меню,
    п_техн.Фамилия + ' ' + п_техн.Имя as Технолог,
    тр.Статус_требования,
    д.Статус as Статус_документа,
    д.Комментарий,
    ISNULL(п_клад.Фамилия + ' ' + п_клад.Имя, 'Не назначен') as Кладовщик,
    тр.Дата_обработки,
    ISNULL((SELECT SUM(Количество) FROM Строки_документов WHERE Документ_id = д.id), 0) as Общее_количество,
    ISNULL((SELECT SUM(Сумма) FROM Строки_документов WHERE Документ_id = д.id), 0) as Общая_сумма
FROM Требования_накладные тр
JOIN Документы д ON тр.Документ_id = д.id
JOIN Меню_на_день м ON тр.Меню_id = м.id
JOIN Пользователи п_техн ON тр.Технолог_id = п_техн.id
LEFT JOIN Пользователи п_клад ON тр.Кладовщик_id = п_клад.id;

CREATE VIEW [dbo].[vw_Калькуляционные_карточки_полные] AS
SELECT 
    кк.Номер,
    б.Наименование as Блюдо,
    тк.Выход as [Выход_порции_г],
    кк.Дата_составления,

    кк.Цена_реализации,

        WHEN кк.Food_Cost <= 25 THEN 'Высокомаржинальное'
        WHEN кк.Food_Cost <= 35 THEN 'Среднемаржинальное'
        ELSE 'Низкомаржинальное'
    END as [Маржинальность],

    ISNULL(u.Фамилия + ' ' + LEFT(u.Имя, 1) + '.', 'Не утверждена') as [Кто_утвердил],
    тк.Номер as [Тех_карта]
FROM Калькуляционные_карточки кк
JOIN Технологические_карты тк ON кк.Технологическая_карта_id = тк.id
JOIN Блюда б ON тк.Блюдо_id = б.id
LEFT JOIN Пользователи u ON кк.Кто_утвердил_id = u.id;

-- 2. Представление меню с ценами и Food Cost
CREATE VIEW [dbo].[vw_Меню_с_ценами] AS
SELECT 
    ROW_NUMBER() OVER (ORDER BY м.Дата, см.Порядок_подачи) as [№п/п],
    м.Дата,
    б.Наименование as Блюдо,
    вб.Наименование as [Вид_блюда],
    кк.Цена_реализации as Цена,
    кк.Food_Cost as [Фудкост_процент],
    см.Количество_порций,
    ISNULL(см.Выход_на_порцию, тк.Выход) as [Выход_г],
    (ISNULL(кк.Цена_реализации, 0) * см.Количество_порций) as [Общая_выручка],
    б.Калорийность_расчетная as [Калорийность_на_порцию],
    м.Статус as [Статус_меню]
FROM Меню_на_день м
JOIN Строки_меню см ON м.id = см.Меню_id
JOIN Блюда б ON см.Блюдо_id = б.id
JOIN Виды_блюд вб ON б.Вид_блюда_id = вб.id
LEFT JOIN Технологические_карты тк ON б.id = тк.Блюдо_id AND тк.Статус = 'Утверждена'
LEFT JOIN Калькуляционные_карточки кк ON тк.id = кк.Технологическая_карта_id AND кк.Статус = 'Утверждена';
GO

-- 3. Представление рентабельности блюд
CREATE VIEW [dbo].[vw_Рентабельность_блюд] AS
SELECT 
    б.Наименование as Блюдо,
    тк.Выход as [Выход_порции_г],
    кк.Номер as [Кальк_карта],
    кк.Себестоимость,
    кк.Цена_реализации,
    кк.Food_Cost as [Фудкост_процент],
    кк.Процент_наценки,
    (кк.Цена_реализации - кк.Себестоимость) as [Прибыль_на_порцию],
    ROUND(CASE 
        WHEN кк.Цена_реализации > 0 
        THEN ((кк.Цена_реализации - кк.Себестоимость) / кк.Цена_реализации) * 100 
        ELSE 0 
    END, 2) as [Маржа_процент],
    CASE 
        WHEN кк.Food_Cost <= 25 THEN 'Высокая'
        WHEN кк.Food_Cost <= 35 THEN 'Средняя'
        ELSE 'Низкая'
    END as [Оценка_рентабельности]
FROM Калькуляционные_карточки кк
JOIN Технологические_карты тк ON кк.Технологическая_карта_id = тк.id
JOIN Блюда б ON тк.Блюдо_id = б.id
WHERE кк.Статус = 'Утверждена';
GO

-- 5. Представление для анализа себестоимости с учетом выхода
CREATE VIEW [dbo].[vw_Анализ_себестоимости] AS
SELECT 
    б.Наименование as Блюдо,
    тк.Выход as [Выход_порции_г],
    кк.Себестоимость as [Себестоимость_общая],
    ROUND(кк.Себестоимость / тк.Выход * 1000, 2) as [Себестоимость_за_кг],
    COUNT(ск.id) as [Количество_ингредиентов],
    MAX(ск.Сумма) as [Макс_стоимость_ингредиента],
    MIN(ск.Сумма) as [Мин_стоимость_ингредиента],
    AVG(ск.Сумма) as [Сред_стоимость_ингредиента]
FROM Калькуляционные_карточки кк
JOIN Технологические_карты тк ON кк.Технологическая_карта_id = тк.id
JOIN Блюда б ON тк.Блюдо_id = б.id
JOIN Строки_калькуляции ск ON кк.id = ск.Калькуляционная_карточка_id
WHERE кк.Статус = 'Утверждена'
GROUP BY б.Наименование, тк.Выход, кк.Себестоимость, кк.id;
GO
